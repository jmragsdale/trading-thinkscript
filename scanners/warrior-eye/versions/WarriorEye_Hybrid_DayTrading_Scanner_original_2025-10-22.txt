# WARRIOR EYE HYBRID - Day Trading Scanner
# Combines Warrior Trading momentum criteria with Eye Pattern timing
# Strategy: Find hot stocks, enter on intraday pullbacks with precision
# Timeframe: Intraday (1min, 5min charts recommended)

# ======================================
# PHASE 1: WARRIOR MOMENTUM PRE-FILTER
# ======================================

# === WARRIOR INPUTS ===
input minPercentGain = 10.0;           # Stock must be up 10% minimum
input minRelativeVolume = 3.0;         # 3x+ relative volume (lowered from 5x for more results)
input minPrice = 1.00;                 # Minimum price $1.00
input maxPrice = 20.00;                # Maximum price $20.00
input maxFloat = 50000000;             # Max 50M float (adjust based on data availability)
input volumeLookbackDays = 30;         # 30-day average volume calculation

# === EYE PATTERN INPUTS ===
input fastMA = 9;                      # Faster MA for intraday (9 period)
input slowMA = 20;                     # Slower MA (20 period)
input pullbackBars = 10;               # Max bars in pullback
input volumeContractionPct = 50;       # Volume must drop 50% during pullback
input minRiskReward = 2.0;             # Minimum 2:1 reward/risk (Warrior standard)

# ======================================
# WARRIOR MOMENTUM CRITERIA
# ======================================

# 1. Price Range Check
def priceInRange = close >= minPrice and close <= maxPrice;

# 2. Percentage Gain Today
# Use GetYYYYMMDD() to identify first bar of the day (works on all timeframes)
def today = GetYYYYMMDD();
def isNewDay = today != today[1];
def todayOpen = if isNewDay then open else todayOpen[1];
def percentChange = if todayOpen > 0 then ((close - todayOpen) / todayOpen) * 100 else 0;
def isUpEnough = percentChange >= minPercentGain;

# 3. Relative Volume Calculation
def avgVolume = Average(volume, volumeLookbackDays);
def relativeVolume = if avgVolume > 0 then volume / avgVolume else 0;
def hasHighVolume = relativeVolume >= minRelativeVolume;

# 4. Volatility Check (using ATR)
def atr = Average(TrueRange(high, close, low), 14);
def atrPercent = (atr / close) * 100;
def isVolatile = atrPercent >= 3; # At least 3% daily range

# === WARRIOR FILTER PASSED ===
def warriorFilter = priceInRange and isUpEnough and hasHighVolume and isVolatile;

# ======================================
# PHASE 2: EYE PATTERN ENTRY TIMING
# ======================================

# Moving Averages (intraday optimized)
def MA_Fast = SimpleMovingAvg(close, fastMA);
def MA_Slow = SimpleMovingAvg(close, slowMA);

# Trend Confirmation (both MAs rising)
def MA_Slow_Rising = MA_Slow > MA_Slow[1];
def MA_Fast_Rising = MA_Fast > MA_Fast[1];
def uptrend = MA_Slow_Rising;

# Crossover Detection
def bullishCross = MA_Fast[1] <= MA_Slow[1] and MA_Fast > MA_Slow;
def bearishCross = MA_Fast[1] >= MA_Slow[1] and MA_Fast < MA_Slow;

# Pullback Counter
def barsInPullback;
if bearishCross {
    barsInPullback = 1;
} else if barsInPullback[1] > 0 and MA_Fast < MA_Slow {
    barsInPullback = barsInPullback[1] + 1;
} else {
    barsInPullback = 0;
}

def validPullbackDuration = barsInPullback > 0 and barsInPullback <= pullbackBars;

# Volume Contraction During Pullback
def maxRecentVolume = Highest(volume, pullbackBars);
def volumeContracted = volume <= maxRecentVolume * (volumeContractionPct / 100.0);

# Lower Shadow Detection (Bullish Reversal Signal)
def bodyLow = Min(open, close);
def bodyHigh = Max(open, close);
def lowerShadowSize = bodyLow - low;
def upperShadowSize = high - bodyHigh;
def bodySize = AbsValue(close - open);
def hasLowerShadow = lowerShadowSize > bodySize * 0.5;
def isBullishCandle = close > open;

# Pullback Setup Forming
def pullbackSetup =
    warriorFilter and
    uptrend and
    validPullbackDuration and
    MA_Fast < MA_Slow and
    volumeContracted;

# ======================================
# BUY SIGNAL (Entry Trigger)
# ======================================

def buySignal =
    pullbackSetup[1] and
    bullishCross and
    isBullishCandle and
    hasLowerShadow;

# Alternative Buy Signal (aggressive - no lower shadow required)
def buySignalAggressive =
    pullbackSetup[1] and
    bullishCross and
    isBullishCandle;

# ======================================
# RISK MANAGEMENT & EXITS
# ======================================

# Stop Loss: Below recent swing low OR below lower shadow
def swingLow = Lowest(low, pullbackBars);
def stopLoss = Min(swingLow, low - lowerShadowSize);
def riskPerShare = close - stopLoss;

# Profit Targets
def target1 = close + (riskPerShare * 2.0);  # 2:1 ratio (Warrior standard)
def target2 = close + (riskPerShare * 3.0);  # 3:1 ratio (extended)
def target3 = close + (riskPerShare * 5.0);  # 5:1 ratio (runner)

# Risk/Reward Ratio
def rrRatio = if riskPerShare > 0 then (target1 - close) / riskPerShare else 0;
def validRiskReward = rrRatio >= minRiskReward;

# Exit Signals
def exitBearishCross = MA_Fast < MA_Slow; # Fast cross below slow = exit
def exitBelowMA20 = close < MA_Slow;       # Close below MA20 = exit
def exitLargeRedCandle = close < open and bodySize > atr * 0.5; # Large red candle

# ======================================
# PATTERN QUALITY SCORE
# ======================================

def qualityScore =
    (if warriorFilter then 3 else 0) +
    (if hasHighVolume and relativeVolume >= 5 then 2 else if hasHighVolume then 1 else 0) +
    (if percentChange >= 20 then 2 else if percentChange >= 10 then 1 else 0) +
    (if volumeContracted then 2 else 0) +
    (if hasLowerShadow then 2 else 0) +
    (if validRiskReward then 2 else 0) +
    (if uptrend then 1 else 0) +
    (if barsInPullback <= 5 then 1 else 0); # Shorter pullback = better

# Max score = 15 (excellent setup)
