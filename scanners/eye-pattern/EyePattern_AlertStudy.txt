# === INPUTS ===
input fastLength = 5;
input slowLength = 20;
input volumeContractionPercent = 30;
input volumeLookback = 20;
input maxPullbackPercent = 15;
input maxPullbackDays = 30;
input ma20SlopeMinimum = 0.0001;
input alertsOn = yes;

# === MOVING AVERAGES ===
def MA5 = SimpleMovingAvg(close, fastLength);
def MA20 = SimpleMovingAvg(close, slowLength);

# === PATTERN DETECTION ===

# 1. MA20 must be rising
def MA20_Slope = (MA20 - MA20[1]) / MA20[1];
def MA20_Rising = MA20_Slope > ma20SlopeMinimum;

# 2. Detect crossovers
def MA5_CrossedBelow = MA5[1] >= MA20[1] and MA5 < MA20;
def MA5_CrossedAbove = MA5[1] <= MA20[1] and MA5 > MA20;

# 3. Volume analysis
def maxVolume = Highest(volume, volumeLookback);
def volumeContracted = volume <= maxVolume * (volumeContractionPercent / 100);

# 4. Pullback validation
def recentHigh = Highest(high, maxPullbackDays);
def pullbackPercent = ((recentHigh - close) / recentHigh) * 100;
def validPullbackSignal = pullbackPercent <= maxPullbackPercent;

# 5. Count days in pullback
def daysSinceCrossBelow;
if MA5_CrossedBelow {
    daysSinceCrossBelow = 1;
} else if daysSinceCrossBelow[1] > 0 and MA5 < MA20 {
    daysSinceCrossBelow = daysSinceCrossBelow[1] + 1;
} else {
    daysSinceCrossBelow = 0;
}

def withinTimeLimit = daysSinceCrossBelow <= maxPullbackDays and daysSinceCrossBelow > 0;

# 6. Lower shadow detection
def bodyLow = Min(open, close);
def lowerShadowSize = bodyLow - low;
def bodySize = AbsValue(close - open);
def hasLowerShadowSignal = lowerShadowSize > bodySize * 0.5;

# === ENTRY SIGNAL (The "Eye" Pattern) ===
def eyePatternForming =
    daysSinceCrossBelow[1] > 0 and
    daysSinceCrossBelow[1] <= maxPullbackDays and
    MA5 >= MA20 and
    MA20_Rising and
    volumeContracted and
    validPullbackSignal;

def entrySignal = eyePatternForming and MA5_CrossedAbove;

# === STOP LOSS & RISK/REWARD ===
def swingLow = Lowest(low, 5);
def stopLoss = Min(swingLow, low - lowerShadowSize * 0.5);
def riskAmount = close - stopLoss;
def target1 = close + (riskAmount * 1.5);
def riskRewardRatio = if riskAmount > 0 then (target1 - close) / riskAmount else 0;

# === PATTERN QUALITY SCORE ===
def patternScore =
    (if MA5 > MA20 then 2 else 0) +
    (if MA20_Rising then 2 else 0) +
    (if volumeContracted then 2 else 0) +
    (if validPullbackSignal then 2 else 0) +
    (if hasLowerShadowSignal then 1 else 0) +
    (if withinTimeLimit then 1 else 0);

# === SCANNER OUTPUT ===
# NOTE: ThinkScript scanners can only have ONE plot statement active at a time.
# Uncomment the plot you want to use for scanning, and comment out the others.

# OPTION 1: BUY NOW Signal (1 = immediate entry)
plot signal = entrySignal;

Alert(
    alertsOn and entrySignal and !entrySignal[1],
    "Eye Pattern: bullish recovery entry signal",
    Alert.BAR,
    Sound.Chimes
);
